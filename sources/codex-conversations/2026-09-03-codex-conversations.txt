For an actual DAM heatmap-shaped table:

```sql
WITH prices AS (
  SELECT
    CAST(
      with_timezone(CAST(application_date AS timestamp), 'UTC')
        AT TIME ZONE 'Asia/Kolkata'
      AS timestamp
    ) AS time_ist,
    value / 1000 AS price_rupees_per_kwh
  FROM india_iex_tsdb_expanded_historical
  WHERE level = 'national'
    AND technology = 'total'
    AND metric = 'price'
    AND market = 'DAM'
    AND value IS NOT NULL
),
bucketed AS (
  SELECT
    year(time_ist) AS delivery_year,
    month(time_ist) AS delivery_month,
    CAST(floor(hour(time_ist) / 4) AS integer) AS band,
    price_rupees_per_kwh
  FROM prices
)
SELECT
  delivery_year,
  CASE band
    WHEN 0 THEN '00:00 - 04:00'
    WHEN 1 THEN '04:00 - 08:00'
    WHEN 2 THEN '08:00 - 12:00'
    WHEN 3 THEN '12:00 - 16:00'
    WHEN 4 THEN '16:00 - 20:00'
    WHEN 5 THEN '20:00 - 24:00'
  END AS time_band,
  round(avg(CASE WHEN delivery_month = 1  THEN price_rupees_per_kwh END), 2) AS jan,
  round(avg(CASE WHEN delivery_month = 2  THEN price_rupees_per_kwh END), 2) AS feb,
  round(avg(CASE WHEN delivery_month = 3  THEN price_rupees_per_kwh END), 2) AS mar,
  round(avg(CASE WHEN delivery_month = 4  THEN price_rupees_per_kwh END), 2) AS apr,
  round(avg(CASE WHEN delivery_month = 5  THEN price_rupees_per_kwh END), 2) AS may,
  round(avg(CASE WHEN delivery_month = 6  THEN price_rupees_per_kwh END), 2) AS jun,
  round(avg(CASE WHEN delivery_month = 7  THEN price_rupees_per_kwh END), 2) AS jul,
  round(avg(CASE WHEN delivery_month = 8  THEN price_rupees_per_kwh END), 2) AS aug,
  round(avg(CASE WHEN delivery_month = 9  THEN price_rupees_per_kwh END), 2) AS sep,
  round(avg(CASE WHEN delivery_month = 10 THEN price_rupees_per_kwh END), 2) AS oct,
  round(avg(CASE WHEN delivery_month = 11 THEN price_rupees_per_kwh END), 2) AS nov,
  round(avg(CASE WHEN delivery_month = 12 THEN price_rupees_per_kwh END), 2) AS dec
FROM bucketed
GROUP BY delivery_year, band
ORDER BY delivery_year, band
```

That should produce six time-band rows per year with months as columns, closely matching Adrien’s example.

For an actual DAM heatmap-shaped table:

```sql
WITH prices AS (
  SELECT
    CAST(
      with_timezone(CAST(application_date AS timestamp), 'UTC')
        AT TIME ZONE 'Asia/Kolkata'
      AS timestamp
    ) AS time_ist,
    value / 1000 AS price_rupees_per_kwh
  FROM india_iex_tsdb_expanded_historical
  WHERE level = 'national'
    AND technology = 'total'
    AND metric = 'price'
    AND market = 'DAM'
    AND value IS NOT NULL
),
bucketed AS (
  SELECT
    year(time_ist) AS delivery_year,
    month(time_ist) AS delivery_month,
    CAST(floor(hour(time_ist) / 4) AS integer) AS band,
    price_rupees_per_kwh
  FROM prices
)
SELECT
  delivery_year,
  CASE band
    WHEN 0 THEN '00:00 - 04:00'
    WHEN 1 THEN '04:00 - 08:00'
    WHEN 2 THEN '08:00 - 12:00'
    WHEN 3 THEN '12:00 - 16:00'
    WHEN 4 THEN '16:00 - 20:00'
    WHEN 5 THEN '20:00 - 24:00'
  END AS time_band,
  round(avg(CASE WHEN delivery_month = 1  THEN price_rupees_per_kwh END), 2) AS jan,
  round(avg(CASE WHEN delivery_month = 2  THEN price_rupees_per_kwh END), 2) AS feb,
  round(avg(CASE WHEN delivery_month = 3  THEN price_rupees_per_kwh END), 2) AS mar,
  round(avg(CASE WHEN delivery_month = 4  THEN price_rupees_per_kwh END), 2) AS apr,
  round(avg(CASE WHEN delivery_month = 5  THEN price_rupees_per_kwh END), 2) AS may,
  round(avg(CASE WHEN delivery_month = 6  THEN price_rupees_per_kwh END), 2) AS jun,
  round(avg(CASE WHEN delivery_month = 7  THEN price_rupees_per_kwh END), 2) AS jul,
  round(avg(CASE WHEN delivery_month = 8  THEN price_rupees_per_kwh END), 2) AS aug,
  round(avg(CASE WHEN delivery_month = 9  THEN price_rupees_per_kwh END), 2) AS sep,
  round(avg(CASE WHEN delivery_month = 10 THEN price_rupees_per_kwh END), 2) AS oct,
  round(avg(CASE WHEN delivery_month = 11 THEN price_rupees_per_kwh END), 2) AS nov,
  round(avg(CASE WHEN delivery_month = 12 THEN price_rupees_per_kwh END), 2) AS dec
FROM bucketed
GROUP BY delivery_year, band
ORDER BY delivery_year, band
```

That should produce six time-band rows per year with months as columns, closely matching Adrien’s example.
